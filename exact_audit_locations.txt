=== FEATURE ROUTE POLICY ===
     9	/**
    10	 * Per-route enablement for the 2026-08-26 increment.
    11	 *
    12	 * This replaces the all-or-nothing [SAFE_SHELL_MODE] switch. Lounge stays the
    13	 * cold-launch surface. Text and list routes are live. The procedural 3D Canvas
    14	 * is reachable only after an explicit safe-scene load and is capped to a
    15	 * bounded component set.
    16	 */
    17	object FeatureRoutePolicy {
    18	    const val SAFE_SCENE_COMPONENT_CAP = 8
    19
    20	    fun isEnabled(tab: MainTab): Boolean = when (tab) {
    21	        MainTab.LOUNGE -> true
    22	        MainTab.REPAIR_MANUAL -> true
    23	        MainTab.MAINTENANCE -> true
    24	        MainTab.DIAGNOSTICS -> true
    25	        MainTab.PARTS_CART -> true
    26	        MainTab.VIEW_3D -> true
    27	    }
    28
    29	    fun requiresFeatureData(tab: MainTab): Boolean = when (tab) {
    30	        MainTab.LOUNGE -> false
    31	        MainTab.PARTS_CART -> false
    32	        MainTab.VIEW_3D -> false
    33	        MainTab.REPAIR_MANUAL -> true
    34	        MainTab.MAINTENANCE -> true
    35	        MainTab.DIAGNOSTICS -> false
    36	    }
    37
    38	    /**
    39	     * First interactive 3D frame must not receive the full 56-part catalog.
    40	     * ALL filter: selected component only. System filter: up to eight parts.
    41	     */
    42	    fun boundedSceneComponents(
    43	        allComponents: List<Component3DModel> = PublishedSportTracData.components,
    44	        selectedComponent: Component3DModel?,
    45	        activeSystem: VehicleSystem
    46	    ): List<Component3DModel> {
    47	        if (activeSystem != VehicleSystem.ALL) {
    48	            return allComponents
    49	                .filter { it.system == activeSystem }
    50	                .take(SAFE_SCENE_COMPONENT_CAP)
    51	        }
    52	        val focused = selectedComponent ?: allComponents.firstOrNull()
    53	        return listOfNotNull(focused)
    54	    }
    55	}
